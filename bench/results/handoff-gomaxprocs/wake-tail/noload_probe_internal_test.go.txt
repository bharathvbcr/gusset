package gusset

import (
	"runtime"
	"slices"
	"testing"
	"time"
)

// The probe's sleeper with a handle open and no Calls: the baseline the
// bystander's lateness is compared against.
func TestProbe_SleeperWithoutCalls(t *testing.T) {
	defer runtime.GOMAXPROCS(runtime.GOMAXPROCS(2))
	h, err := Open(WithPoolSize(2), WithDiagnosticEngine())
	if err != nil {
		t.Fatal(err)
	}
	defer h.Close()
	const nap = 200 * time.Microsecond
	var late []time.Duration
	time.Sleep(20 * time.Millisecond)
	for deadline := time.Now().Add(2 * time.Second); len(late) < 1000 && time.Now().Before(deadline); {
		t0 := time.Now()
		time.Sleep(nap)
		late = append(late, time.Since(t0)-nap)
	}
	slices.Sort(late)
	q := func(f float64) time.Duration { return late[int(f*float64(len(late)-1))] }
	var o200, o1ms int
	for _, d := range late {
		if d > nap {
			o200++
		}
		if d > time.Millisecond {
			o1ms++
		}
	}
	t.Logf("PROBE n=%d p25=%v p50=%v p75=%v p90=%v p95=%v p99=%v max=%v over200us=%d over1ms=%d",
		len(late), q(.25), q(.5), q(.75), q(.9), q(.95), q(.99), late[len(late)-1], o200, o1ms)
}
